% --- Plan panels (filters/plan-code.lua) -------------------------------------
% HTML shows the plan and its code side by side; the print text block is too
% narrow for two columns of Python, so the plan is stacked above the code in a
% ruled panel. Kernel LaTeX only — no package dependency to collide with the
% template's own tcolorbox options.
% A short page-room check keeps the plan with the opening of its code when
% possible. The plan itself stays breakable: the component now covers every
% learner-visible listing, including long training and experiment cells.
\newenvironment{planbox}{%
  \par\addvspace{\medskipamount}%
  \begingroup
    \dimen0=\pagegoal \advance\dimen0 by -\pagetotal
    \ifdim\dimen0<9\baselineskip \newpage \fi
  \endgroup
  \begingroup%
  \noindent\rule{\linewidth}{0.6pt}\par\nobreak\vspace{1pt}%
  \noindent{\scshape\footnotesize Plan}\par\nobreak%
  \small%
}{%
  \par\nobreak\vspace{1pt}%
  \noindent\rule{\linewidth}{0.6pt}%
  \endgroup%
  \par\addvspace{\smallskipamount}%
  \begingroup
    \dimen0=\pagegoal \advance\dimen0 by -\pagetotal
    \ifdim\dimen0<8\baselineskip \newpage \fi
  \endgroup
  \noindent{\scshape\footnotesize Code}\par\nobreak%
  \nopagebreak[4]%
}
