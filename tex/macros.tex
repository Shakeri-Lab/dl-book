% Quarto declares custom float counters within the current chapter for scrbook.
% Interlude and epilogue figures own independent EX., AE., TTR., and E. sequences;
% experiment-interlude tables own EX. as well. These match the HTML counters
% rather than inheriting neighboring chapters.
\makeatletter
\AtBeginDocument{%
  \@removefromreset{exfig}{chapter}%
  \@removefromreset{aefig}{chapter}%
  \@removefromreset{ttrfig}{chapter}%
  \@removefromreset{epfig}{chapter}%
  \@removefromreset{extbl}{chapter}%
  \renewcommand{\theexfig}{\arabic{exfig}}%
  \renewcommand{\theaefig}{\arabic{aefig}}%
  \renewcommand{\thettrfig}{\arabic{ttrfig}}%
  \renewcommand{\theepfig}{\arabic{epfig}}%
  \renewcommand{\theextbl}{\arabic{extbl}}%
  % The experiment tables are short but Pandoc emits longtable inside the custom
  % float. A deferred page float can swallow the intervening subsection while it
  % searches for placement. Pin these two audit ledgers here: both fit cleanly,
  % and their neighboring derivation must remain in the PDF reading flow.
  \floatplacement{extbl}{H}%
}
\makeatother

% scrbook defaults to \flushbottom (two-sided); when a figure forces a short
% page, TeX stretched display/verbatim glue into ~2in voids that separated
% code cells from their printed outputs. Collect leftover space at the bottom.
\raggedbottom
% Hyperref needs the anchor for an unnumbered heading *after* the heading command:
% the command may move to the next page. Pandoc writes \label and
% \addcontentsline immediately afterward, so one post-command anchor gives both
% the label and PDF outline the heading's actual page. Wrap only starred headings;
% numbered commands already create their anchors internally.
\AtBeginDocument{%
  \let\dlbookChapter\chapter
  \let\dlbookSection\section
  \let\dlbookSubsection\subsection
  \let\dlbookSubsubsection\subsubsection
  \RenewDocumentCommand{\chapter}{s o m}{%
    \IfBooleanTF{#1}{%
      \IfNoValueTF{#2}{\dlbookChapter*{#3}}{\dlbookChapter*[#2]{#3}}%
      \phantomsection
    }{%
      \IfNoValueTF{#2}{\dlbookChapter{#3}}{\dlbookChapter[#2]{#3}}%
    }%
  }%
  \RenewDocumentCommand{\section}{s o m}{%
    \IfBooleanTF{#1}{%
      \IfNoValueTF{#2}{\dlbookSection*{#3}}{\dlbookSection*[#2]{#3}}%
      \phantomsection
    }{%
      \IfNoValueTF{#2}{\dlbookSection{#3}}{\dlbookSection[#2]{#3}}%
    }%
  }%
  \RenewDocumentCommand{\subsection}{s o m}{%
    \IfBooleanTF{#1}{%
      \IfNoValueTF{#2}{\dlbookSubsection*{#3}}{\dlbookSubsection*[#2]{#3}}%
      \phantomsection
    }{%
      \IfNoValueTF{#2}{\dlbookSubsection{#3}}{\dlbookSubsection[#2]{#3}}%
    }%
  }%
  \RenewDocumentCommand{\subsubsection}{s o m}{%
    \IfBooleanTF{#1}{%
      \IfNoValueTF{#2}{\dlbookSubsubsection*{#3}}{\dlbookSubsubsection*[#2]{#3}}%
      \phantomsection
    }{%
      \IfNoValueTF{#2}{\dlbookSubsubsection{#3}}{\dlbookSubsubsection[#2]{#3}}%
    }%
  }%
}

% PDF is a narrow derived format. Source lines are capped by CI, but frozen stdout
% and identifiers can still be intrinsically long. Wrap them visibly instead of
% allowing TeX to print past the paper edge. HTML remains the canonical source view.
\AtBeginDocument{%
  \RecustomVerbatimEnvironment{Highlighting}{Verbatim}{%
    commandchars=\\\{\},%
    breaklines=true,%
    breakanywhere=true,%
    breakautoindent=true,%
    breakindent=1em%
  }%
  \RecustomVerbatimEnvironment{verbatim}{Verbatim}{%
    breaklines=true,%
    breakanywhere=true,%
    breakautoindent=true,%
    breakindent=1em%
  }%
}

% --- Plan panels (filters/plan-code.lua) -------------------------------------
% HTML shows the plan and its code side by side; the print text block is too
% narrow for two columns of Python, so the plan is stacked above the code in a
% ruled panel. Kernel LaTeX only — no package dependency to collide with the
% template's own tcolorbox options.
% A short page-room check keeps the plan with the opening of its code when
% possible. The plan itself stays breakable: the component now covers every
% learner-visible listing, including long training and experiment cells.
\newenvironment{planbox}{%
  \par\addvspace{\medskipamount}%
  \begingroup
    \dimen0=\pagegoal \advance\dimen0 by -\pagetotal
    \ifdim\dimen0<9\baselineskip \newpage \fi
  \endgroup
  \begingroup%
  \noindent\rule{\linewidth}{0.6pt}\par\nobreak\vspace{1pt}%
  \noindent{\scshape\footnotesize Plan}\par\nobreak%
  \small%
  \let\dlbookplanunderscore\_%
  \renewcommand{\_}{\dlbookplanunderscore\discretionary{}{}{}}%
}{%
  \par\nobreak\vspace{1pt}%
  \noindent\rule{\linewidth}{0.6pt}%
  \endgroup%
  \par\addvspace{\smallskipamount}%
  \begingroup
    \dimen0=\pagegoal \advance\dimen0 by -\pagetotal
    \ifdim\dimen0<8\baselineskip \newpage \fi
  \endgroup
  \noindent{\scshape\footnotesize Code}\par\nobreak%
  \nopagebreak[4]%
}
