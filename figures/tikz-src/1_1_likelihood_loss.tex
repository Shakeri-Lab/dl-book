\begin{tikzpicture}[
  >=Stealth,
  node distance=0.55cm and 0.55cm,
  box/.style={
    draw=gray!50, rounded corners=3pt, minimum height=0.85cm,
    text width=2.45cm, align=center, inner sep=4pt, font=\small
  },
  arrow/.style={-{Stealth}, thick, gray!70}
]
  \node[box, draw=dlpurple] (real)
    {$\targetpart{y}\in\R$\\continuous target};
  \node[box, right=of real] (gauss)
    {$\targetpart{y}\mid\featurepart{x}
      \sim\mathcal N(\predictionpart{\hat y},\sigma^2)$\\identity link};
  \node[box, right=of gauss] (nll1)
    {$-\sum_i\log p(\targetpart{y_i}\mid\featurepart{x_i})$};
  \node[box, right=of nll1] (mse)
    {\textbf{MSE}\\
     $\sum_i(\targetpart{y_i}-\predictionpart{\hat y_i})^2$};

  \node[box, draw=dlpurple, below=0.75cm of real] (binary)
    {$\targetpart{y}\in\{0,1\}$\\binary target};
  \node[box, right=of binary] (bern)
    {$\targetpart{y}\mid\featurepart{x}
      \sim\operatorname{Bernoulli}(\predictionpart{\hat p})$\\sigmoid link};
  \node[box, right=of bern] (nll2)
    {$-\sum_i\log p(\targetpart{y_i}\mid\featurepart{x_i})$};
  \node[box, right=of nll2] (bce)
    {\textbf{binary cross-entropy}\\
     $-\sum_i[\targetpart{y_i}\log\predictionpart{\hat p_i}
       +(1-\targetpart{y_i})\log(1-\predictionpart{\hat p_i})]$};

  \foreach \a/\b in {real/gauss,gauss/nll1,nll1/mse,binary/bern,bern/nll2,nll2/bce}
    \draw[arrow] (\a) -- (\b);

  \node[font=\footnotesize\bfseries, gray!80, left=0.2cm of real] {Chapter 1};
  \node[font=\footnotesize\bfseries, gray!80, left=0.2cm of binary] {Chapter 2};
  \node[font=\footnotesize, gray!75, above=0.25cm of nll1]
    {same maximum-likelihood recipe};
\end{tikzpicture}
